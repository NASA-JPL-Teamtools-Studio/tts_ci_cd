16def extract_label_from_record(data): 

17 """Helper to extract and normalize a severity label from a single OSV record.""" 

18 # 1. Try to get the human-readable label (Common for GHSA) 

19 db_specific = data.get("database_specific", {}) 

20 if "severity" in db_specific and isinstance(db_specific["severity"], str): 

21 label = db_specific["severity"].upper() 

22 return "MODERATE" if label == "MEDIUM" else label 

23 

24 # 2. Try to calculate it from a CVSS V3 Vector 

25 for sev in data.get("severity", []): 

26 if sev.get("type") == "CVSS_V3": 

27 vector = sev.get("score") 

28 try: 

29 c = CVSS3(vector) 

30 score = c.base_score 

31 if score >= 9.0: return "CRITICAL" 

32 if score >= 7.0: return "HIGH" 

33 if score >= 4.0: return "MODERATE" 

34 if score > 0.0: return "LOW" 

35 except Exception as e: 

36 print(f"Warning: Failed to parse CVSS vector '{vector}': {e}") 

37 pass 

38 

39 return None

41def get_severity_label(vuln_id): 

42 """Robust lookup that pivots to aliases AND related IDs if the primary record is empty.""" 

43 if vuln_id in SEVERITY_CACHE: 

44 return SEVERITY_CACHE[vuln_id] 

45 

46 label = "UNKNOWN" 

47 base_url = "https://api.osv.dev/v1/vulns/" 

48 

49 try: 

50 # Step 1: Query the primary ID 

51 resp = requests.get(f"{base_url}{vuln_id}", timeout=5) 

52 if resp.status_code == 200: 

53 data = resp.json() 

54 

55 # Extract from the primary record 

56 extracted = extract_label_from_record(data) 

57 if extracted: 

58 SEVERITY_CACHE[vuln_id] = extracted 

59 return extracted 

60 

61 # Step 2: Alias & Related Pivot 

62 # Python wrappers of vulnerable C-libraries (like libwebp) use "related" instead of "aliases" 

63 pivot_ids = data.get("aliases", []) + data.get("related", []) 

64 

65 # Prioritize GitHub Security Advisories, then standard CVEs 

66 best_pivots = [a for a in pivot_ids if a.startswith("GHSA")] + \ 

67 [a for a in pivot_ids if a.startswith("CVE")] 

68 

69 for pivot_id in best_pivots: 

70 pivot_resp = requests.get(f"{base_url}{pivot_id}", timeout=5) 

71 if pivot_resp.status_code == 200: 

72 pivot_data = pivot_resp.json() 

73 extracted = extract_label_from_record(pivot_data) 

74 if extracted: 

75 SEVERITY_CACHE[vuln_id] = extracted 

76 return extracted 

77 

78 except Exception as e: 

79 print(f"Error fetching severity for {vuln_id}: {e}") 

80 

81 # Step 3: Fallback 

82 SEVERITY_CACHE[vuln_id] = label 

83 return label
