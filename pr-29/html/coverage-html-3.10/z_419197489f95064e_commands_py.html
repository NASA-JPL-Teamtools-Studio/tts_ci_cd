97def run_class(db: IndexDB, class_path: str, depth: int = 1): 

98 """Inspect a class and its MRO.""" 

99 cls = db.get_class_by_path(class_path) 

100 if not cls: 

101 print(f"Class not found: {class_path}") 

102 return 

103 

104 mod = db.get_module_by_id(cls['module_id']) 

105 repo_name = mod['repo_name'] if mod else "?" 

106 

107 print(f"## {cls['full_path']}") 

108 print(f" \"{cls['doc_summary']}\"") 

109 print(f" Repo: {repo_name}") 

110 print(f" Public: {cls['is_public']}\n") 

111 

112 print(" MRO:") 

113 mro = db.get_class_mro(cls['id']) 

114 for i, (cid, path, ext) in enumerate(mro): 

115 indent = " " * i 

116 name = path if path else f"(external: {ext})" 

117 print(f"{indent}├── {name}") 

118 

119 if depth >= 2: 

120 methods = db.get_class_methods(cls['id']) 

121 print("\nMethods:") 

122 for m in methods: 

123 summary = (m['doc_summary'] or "").strip().splitlines()[0] if m['doc_summary'] else "" 

124 sig = m['signature'] or "()" 

125 print(f" • {m['name']}{sig}" + (f" - {summary}" if summary else "")) 

126 

127 if depth >= 3: 

128 if mod: 

129 print(f"\nSource ({mod['file_path']}):") 

130 file_path = Path(mod['file_path']) 

131 if file_path.exists(): 

132 lines = file_path.read_text(encoding="utf-8").splitlines() 

133 start = max(cls['line_start'] - 1, 0) 

134 end = min(cls['line_end'], len(lines)) 

135 for lineno, line in enumerate(lines[start:end], start=start + 1): 

136 print(f" {lineno:>5} | {line}") 

137 else: 

138 print(f" [File not found: {file_path}]")

161def run_deps(db: IndexDB, start_repo: str): 

162 """Truthful dependency mapping for a library, with mission-aware filtering.""" 

163 from tts_ci_cd.ctx.config import get_current_mission 

164 current_mission = get_current_mission() 

165 

166 repo = db.get_repo_by_name(start_repo) 

167 if not repo: 

168 print(f"Repo not found: {start_repo}") 

169 return 

170 

171 print(f"Dependency Graph for {start_repo}") 

172 if current_mission: 

173 print(f"Filtering for mission: {current_mission}") 

174 print("-" * 40) 

175 

176 visited = set() 

177 # (repo_name, depth, prefix) 

178 stack = [(start_repo, 0, "")] 

179 

180 STD_LIB = {"sys", "os", "re", "json", "datetime", "collections", "pathlib", "logging", "typing", "abc", "copy", "enum", "functools", "itertools", "math", "random", "time", "traceback", "uuid", "tempfile", "inspect", "importlib", "argparse", "shutil", "subprocess", "xml.etree.ElementTree"} 

181 

182 while stack: 

183 current_repo, depth, prefix = stack.pop() 

184 

185 # Resolve tagging for the current node 

186 repo_info = db.get_repo_by_name(current_repo) 

187 tag = "" 

188 if repo_info: 

189 if repo_info['type'] == "adaptation": 

190 tag = " [ADAPTATION]" 

191 elif repo_info['type'] == "core": 

192 tag = " [CORE]" 

193 

194 if current_repo == start_repo: 

195 print(f"{start_repo}{tag}") 

196 else: 

197 print(f"{prefix}└── {current_repo}{tag}") 

198 

199 if current_repo in visited: 

200 continue 

201 visited.add(current_repo) 

202 

203 modules = db.conn.execute( 

204 "SELECT id FROM modules WHERE repo_name = ?", (current_repo,) 

205 ).fetchall() 

206 

207 deps_for_this_repo = set() 

208 

209 for mod in modules: 

210 imports = db.get_imports_for_module(mod['id']) 

211 for imp in imports: 

212 res_mod = imp['resolved_module'] 

213 if not res_mod: 

214 continue 

215 

216 repo_row = db.conn.execute( 

217 "SELECT repo_name FROM modules WHERE dotted_path = ? OR dotted_path LIKE ? LIMIT 1", 

218 (res_mod, f"{res_mod}.%") 

219 ).fetchone() 

220 

221 if repo_row: 

222 dep_repo_name = repo_row['repo_name'] 

223 if dep_repo_name != current_repo: 

224 # Mission Filtering 

225 dep_repo_info = db.get_repo_by_name(dep_repo_name) 

226 if dep_repo_info: 

227 mission = dep_repo_info['mission'] 

228 # Keep if it's core, matches current mission, or is the root 

229 if not mission or mission == current_mission or dep_repo_name == start_repo: 

230 deps_for_this_repo.add(dep_repo_name) 

231 else: 

232 deps_for_this_repo.add(dep_repo_name) 

233 else: 

234 base_mod = res_mod.split('.')[0] 

235 if base_mod not in STD_LIB: 

236 deps_for_this_repo.add(f"(external: {res_mod})") 

237 

238 for dep in sorted(list(deps_for_this_repo)): 

239 if dep.startswith("(external:"): 

240 print(f"{prefix} └── {dep}") 

241 elif dep != start_repo: 

242 stack.append((dep, depth + 1, prefix + " "))
