1""" 

2Validate agent pointer conventions across TTS repos. 

3 

4Checks: 

5- docs/agents/README.md exists for in-scope repos 

6- File contains required 5 fields: Issue tracker, Adapts, Triage labels, Domain context, Ownership 

7- No legacy docs/agents/{issue-tracker,domain,triage-labels}.md remain 

8- ## Agent skills block not present in AGENTS.md 

9""" 

10 

11from __future__ import annotations 

12 

13import re 

14import sys 

15from pathlib import Path 

16 

17WORKSPACE_ROOT = Path("/Users/muszynsk/projects/tt_studio/dev") 

18 

19REQUIRED_FIELDS = ["Issue tracker:", "Adapts:", "Triage labels:", "Domain context:", "Ownership:"] 

20 

21def is_in_scope(rel: str) -> bool: 

22 name = Path(rel).name 

23 if re.match(r"^tts_[a-z0-9_]+$", name): 

24 return True 

25 m = re.match(r"^([a-z0-9_]+)_([a-z0-9_]+)$", name) 

26 if m: 

27 mission, lib = m.groups() 

28 core_name = f"tts_{lib}" 

29 # check if core exists in workspace 

30 # simplistic: look for tts_core/tts_lib 

31 if (WORKSPACE_ROOT / "tts_core" / core_name).exists(): 

32 return True 

33 return False 

34 

35def validate_repo(repo_path: Path, rel: str) -> list[str]: 

36 issues = [] 

37 agents_dir = repo_path / "docs" / "agents" 

38 readme = agents_dir / "README.md" 

39 if not readme.exists(): 

40 issues.append(f"Missing docs/agents/README.md") 

41 return issues 

42 content = readme.read_text() 

43 for field in REQUIRED_FIELDS: 

44 if field not in content: 

45 issues.append(f"Missing field '{field}' in README.md") 

46 # check legacy files 

47 for legacy in ["issue-tracker.md", "domain.md", "triage-labels.md"]: 

48 if (agents_dir / legacy).exists(): 

49 issues.append(f"Legacy file still present: {legacy}") 

50 # check AGENTS.md for ## Agent skills block 

51 agents_md = repo_path / "AGENTS.md" 

52 if agents_md.exists(): 

53 txt = agents_md.read_text() 

54 if re.search(r"^## Agent skills", txt, re.M): 

55 issues.append("AGENTS.md still contains ## Agent skills block") 

56 return issues

58def main(argv: list[str] | None = None): 

59 repos = [] 

60 for root, dirs, _ in Path(WORKSPACE_ROOT).walk():

62 repo_path = Path(root) 

63 rel = str(repo_path.relative_to(WORKSPACE_ROOT)) 

64 if is_in_scope(rel):
